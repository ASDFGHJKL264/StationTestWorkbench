cmake_minimum_required(VERSION 3.19)
project(StationTestWorkbench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Sql Widgets)
include(CTest)

add_library(WorkbenchDomain STATIC src/core/domain.h src/core/domain.cpp)
target_include_directories(WorkbenchDomain PUBLIC src)
target_link_libraries(WorkbenchDomain PUBLIC Qt6::Core)
add_library(WorkbenchProtocol STATIC src/protocol/framecodec.h src/protocol/framecodec.cpp)
target_include_directories(WorkbenchProtocol PUBLIC src)
target_link_libraries(WorkbenchProtocol PUBLIC Qt6::Core)
add_library(WorkbenchStorage STATIC src/storage/repository.h src/storage/repository.cpp)
target_link_libraries(WorkbenchStorage PUBLIC WorkbenchDomain Qt6::Sql)
add_library(WorkbenchRuntime STATIC src/runtime/testengine.h src/runtime/testengine.cpp
    src/runtime/workbenchcontroller.h src/runtime/workbenchcontroller.cpp
    src/runtime/asyncfilelogger.h src/runtime/asyncfilelogger.cpp)
target_link_libraries(WorkbenchRuntime PUBLIC WorkbenchDomain WorkbenchStorage WorkbenchProtocol Qt6::Network)
add_library(WorkbenchWidgets STATIC src/ui/toolstyle.h src/ui/mainwindow.h src/ui/mainwindow.cpp src/ui/models.h src/ui/models.cpp)
target_link_libraries(WorkbenchWidgets PUBLIC WorkbenchRuntime Qt6::Widgets)
add_executable(StationTestWorkbench WIN32 src/main.cpp)
target_link_libraries(StationTestWorkbench PRIVATE WorkbenchWidgets)
add_library(StationSimulation STATIC src/simulator/stationserver.h src/simulator/stationserver.cpp)
target_link_libraries(StationSimulation PUBLIC WorkbenchDomain WorkbenchProtocol Qt6::Network)
add_library(StationSimulationWidgets STATIC
    src/simulator/simulatorcontroller.h src/simulator/simulatorcontroller.cpp
    src/simulator/simulatorwindow.h src/simulator/simulatorwindow.cpp)
target_link_libraries(StationSimulationWidgets PUBLIC StationSimulation Qt6::Widgets)
add_executable(StationSimulator WIN32 src/simulator/main.cpp)
target_link_libraries(StationSimulator PRIVATE StationSimulationWidgets)

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    function(workbench_test name file)
        add_executable(${name} ${file})
        target_link_libraries(${name} PRIVATE ${ARGN} Qt6::Test)
        add_test(NAME ${name} COMMAND ${name} -o ${name}-results.txt,txt)
        set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    endfunction()
    workbench_test(ProtocolTests tests/test_protocol.cpp WorkbenchProtocol)
    workbench_test(DomainTests tests/test_domain.cpp WorkbenchDomain)
    workbench_test(RepositoryTests tests/test_repository.cpp WorkbenchStorage)
    workbench_test(ModelTests tests/test_models.cpp WorkbenchWidgets)
    workbench_test(WorkflowTests tests/test_workflow.cpp WorkbenchRuntime StationSimulation)
    workbench_test(WindowTests tests/test_window.cpp WorkbenchWidgets StationSimulation)
    workbench_test(SimulatorTests tests/test_simulator.cpp StationSimulation)
    workbench_test(SimulatorWindowTests tests/test_simulator_window.cpp StationSimulationWidgets WorkbenchWidgets)
    workbench_test(LoggerTests tests/test_logger.cpp WorkbenchRuntime)
    workbench_test(ProcessTests tests/test_process.cpp WorkbenchRuntime)
    add_dependencies(ProcessTests StationSimulator)
    add_executable(HistoryBenchmark tests/benchmark_history.cpp)
    target_link_libraries(HistoryBenchmark PRIVATE WorkbenchStorage)
    if(WIN32)
        target_link_libraries(HistoryBenchmark PRIVATE psapi)
    endif()
    set_tests_properties(WorkflowTests PROPERTIES TIMEOUT 900)
    set_tests_properties(ModelTests WindowTests SimulatorWindowTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(WIN32)
        file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" systemFonts)
        set_tests_properties(ModelTests WindowTests SimulatorWindowTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=${systemFonts}")
    endif()
endif()
install(TARGETS StationTestWorkbench StationSimulator RUNTIME DESTINATION .)
